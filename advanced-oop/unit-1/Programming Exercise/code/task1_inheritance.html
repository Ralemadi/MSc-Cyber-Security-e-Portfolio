class Vehicle:
    """Represent a general vehicle."""

    def __init__(self, brand: str, fuel_type: str) -> None:
        self.brand = brand
        self.fuel_type = fuel_type

    def display_details(self) -> str:
        return f"Brand: {self.brand}, Fuel type: {self.fuel_type}"


class Car(Vehicle):
    """Represent a car that inherits from Vehicle."""

    def __init__(self, brand: str, fuel_type: str, num_doors: int) -> None:
        super().__init__(brand, fuel_type)
        self.num_doors = num_doors

    def display_details(self) -> str:
        return f"{super().display_details()}, Number of doors: {self.num_doors}"


if __name__ == "__main__":
    car = Car("Toyota", "Petrol", 4)
    print(car.display_details())
