from dataclasses import dataclass
import hashlib
import hmac
import re
import secrets
from typing import Dict, Optional


@dataclass(frozen=True)
class User:

    user_id: str
    email: str
    password_hash: str


class UserManagementService:

    EMAIL_PATTERN = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
    MIN_PASSWORD_LENGTH = 12
    HASH_ITERATIONS = 200_000

    def __init__(self) -> None:
        # In-memory storage for the scope of this formative activity.
        self._users: Dict[str, User] = {}

    def register_user(self, email: str, password: str) -> User:

        email = self._normalise_email(email)

        if not self.EMAIL_PATTERN.fullmatch(email):
            raise ValueError("Invalid email address.")

        if not isinstance(password, str) or len(password) < self.MIN_PASSWORD_LENGTH:
            raise ValueError(
                f"Password must contain at least {self.MIN_PASSWORD_LENGTH} characters."
            )

        if email in self._users:
            raise ValueError("User already exists.")

        salt = secrets.token_bytes(16)
        password_hash = self._hash_password(password, salt)

        user = User(
            user_id=secrets.token_hex(8),
            email=email,
            password_hash=password_hash
        )

        self._users[email] = user
        return user

    def authenticate(self, email: str, password: str) -> bool:

        if not isinstance(email, str) or not isinstance(password, str):
            return False

        email = email.strip().lower()
        user = self._users.get(email)

        if user is None:
            return False

        return self._verify_password(
            password,
            user.password_hash
        )

    def get_user(self, email: str) -> Optional[User]:

        if not isinstance(email, str):
            return None

        return self._users.get(
            email.strip().lower()
        )

    def _normalise_email(self, email: str) -> str:

        if not isinstance(email, str):
            raise ValueError("Invalid email address.")

        return email.strip().lower()

    def _hash_password(self, password: str, salt: bytes) -> str:

        digest = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            self.HASH_ITERATIONS
        )

        return f"{salt.hex()}:{digest.hex()}"

    def _verify_password(self, password: str, stored_value: str) -> bool:

        salt_hex, stored_hash = stored_value.split(":", 1)
        salt = bytes.fromhex(salt_hex)

        candidate_hash = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            self.HASH_ITERATIONS
        ).hex()

        return hmac.compare_digest(
            stored_hash,
            candidate_hash
        )
