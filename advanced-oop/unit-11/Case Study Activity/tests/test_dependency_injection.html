import sys
import unittest
from pathlib import Path
from unittest.mock import Mock

CODE_DIR = Path(__file__).resolve().parents[1] / "code"
sys.path.insert(0, str(CODE_DIR))

from dependency_injection import NotificationService, UserManager


class TestUserManager(unittest.TestCase):

    def setUp(self) -> None:
        # A mock replaces the real EmailService.
        self.mock_notifier = Mock(spec=NotificationService)
        self.user_manager = UserManager(self.mock_notifier)

    def test_register_user_sends_welcome_notification(self):
        self.user_manager.register_user("user1@example.com")

        self.mock_notifier.send_notification.assert_called_once_with(
            "user1@example.com",
            "Welcome!"
        )

    def test_register_user_strips_whitespace_before_notification(self):
        self.user_manager.register_user("  user1@example.com  ")

        self.mock_notifier.send_notification.assert_called_once_with(
            "user1@example.com",
            "Welcome!"
        )

    def test_register_user_rejects_empty_user(self):
        with self.assertRaises(ValueError):
            self.user_manager.register_user("")

        self.mock_notifier.send_notification.assert_not_called()

    def test_register_user_rejects_non_string_user(self):
        with self.assertRaises(ValueError):
            self.user_manager.register_user(None)

        self.mock_notifier.send_notification.assert_not_called()


if __name__ == "__main__":
    unittest.main(verbosity=2)
