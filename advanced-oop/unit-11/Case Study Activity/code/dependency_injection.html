from abc import ABC, abstractmethod


class NotificationService(ABC):
    """Defines the notification behaviour required by UserManager."""

    @abstractmethod
    def send_notification(self, user: str, message: str) -> None:
        pass


class EmailService(NotificationService):
    """Sends notifications by email."""

    def send_notification(self, user: str, message: str) -> None:
        print(f"Sending email to {user}: {message}")


class SMSService(NotificationService):
    """Alternative notification service demonstrating extensibility."""

    def send_notification(self, user: str, message: str) -> None:
        print(f"Sending SMS to {user}: {message}")


class UserManager:
    """Registers users and delegates notification delivery."""

    def __init__(self, notifier: NotificationService) -> None:
        # Dependency Injection: the notification service is supplied externally.
        self.notifier = notifier

    def register_user(self, user: str) -> None:
        if not isinstance(user, str) or not user.strip():
            raise ValueError("User must be a non-empty string.")

        user = user.strip()

        # Registration behaviour would normally be implemented here.
        self.notifier.send_notification(user, "Welcome!")


if __name__ == "__main__":
    # Composition root: dependency creation happens outside UserManager.
    email_service = EmailService()
    user_manager = UserManager(email_service)
    user_manager.register_user("user1@example.com")

    # UserManager can use another implementation without changing its code.
    sms_service = SMSService()
    sms_user_manager = UserManager(sms_service)
    sms_user_manager.register_user("user2@example.com")
